# Packages of the test guest. Used by create.sh (cloud-init) and provision.sh (running guest).
# Every entry says why it is there; keep this file and CLAUDE.md "Testing environment" in sync.
bubblewrap            # the sandbox under test
squashfs-tools        # unsquashfs (extraction) and mksquashfs (synthetic test images)
git                   # manifest history (run in a jail), and the repository tooling
python3               # the interpreter appjail and the tests run on
python3-tk            # tkinter for the optional GUI (appjail-gui); optional on a user's machine, needed by tests/test_gui_smoke.py
xvfb                  # X server for the GUI smoke test (tkinter needs a display; the guest has Wayland only)
python3-pytest        # test runner (development dependency only)
reuse                 # `reuse lint`: SPDX/REUSE compliance check
python3-venv          # venv with pip: the packaging test installs appjail the way a user would (pip downloads hatchling)
python3-pip           # installs the pinned ruff (tests/vm/install-ruff.sh): no distribution package
ca-certificates       # TLS roots for the HTTPS tests (GitHub API, downloads)
weston                # headless Wayland compositor and weston-simple-shm test client
sway                  # headless wlroots compositor: offers wp_security_context_v1, which weston 14 does not
linux-image-amd64     # regular kernel: the cloud kernel has no DRM drivers (virtio-gpu, /dev/dri)
libopengl0            # host libraries GUI AppImages expect to find on the system
libgl1
libegl1
libglx-mesa0
libgl1-mesa-dri       # software (llvmpipe) GL/EGL drivers: GUI AppImages in the VM have no GPU
libxkbcommon0
libwayland-client0
linux-libc-dev        # unistd_64.h: the reference the seccomp syscall table is checked against
xdg-dbus-proxy        # D-Bus filtering for permissions.dbus_filter (run by appjail outside the app's jail)
dbus-bin              # dbus-send, to talk to a bus from inside the jail in tests
fontconfig            # font lookup for GUI AppImages (the lagrange end-to-end test)
fonts-dejavu-core     # at least one font, or GUI AppImages abort
xdg-desktop-portal    # frontend and document portal: the real portal that tests/test_portal.py talks to
fuse3                 # the document portal's FUSE mount (fusermount3)
xdg-desktop-portal-gtk # a real portal backend (GTK): tests/test_portal.py runs the portal frontend against it under headless weston
python3-dbus          # fake portal backend and in-jail test client
python3-gi            # GLib main loop for python3-dbus
xwayland              # rootless X server for display = "xwayland" (run inside the app's jail)
x11-utils             # xdpyinfo and xwininfo, the X11 clients used by the Xwayland tests
x11-apps              # xwd, to take screenshots of the GUI under Xvfb
passt                 # pasta: private network namespace for the download, attestation and index jails and for every app with `network` (required there)
librsvg2-bin          # rsvg-convert: SVG menu icons are rendered to PNG inside a jail (tests/test_desktop.py)
desktop-file-utils    # desktop-file-validate and update-desktop-database for the generated menu entries
libglib2.0-bin        # gio: tests/test_e2e.py opens a file through the generated menu entry with `gio launch` / `gio open`
pipewire              # tests/test_audio.py: a private PipeWire, to check that the `audio` permission never exposes its socket
pipewire-bin          # pw-dump: what a client can see, from the host and from the jail (tests/test_audio.py)
pipewire-pulse        # the Pulse server whose socket the `audio` permission binds (tests/test_audio.py)
libspa-0.2-modules    # SPA plugins, including videotestsrc: the camera stand-in of tests/test_audio.py (the guest has no video device)
